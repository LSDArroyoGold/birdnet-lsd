# Parametros del motor de deteccion propio (loop custom, no BirdNET-Pi).
# Mismo formato que config_general.txt / config_horarios.txt del resto del
# proyecto: KEY = valor, editable en caliente sin tocar codigo.

# --- Clasificacion (modelo BirdNET) ---
CONFIDENCE = 0.6
SENSITIVITY = 1.1

# --- Disparador de actividad acustica (NO es el clasificador de especies,
# es un chequeo de señal liviano que corre continuo sobre el audio
# entrante para decidir "esta pasando algo ahi"). Banda encontrada
# empiricamente el 22/08 comparando canto confirmado de 195 especies
# (Xeno-canto) contra ruido de campo real: contraste maximo en banda
# angosta 3500-4500Hz (34.9dB), pero se usa una banda mas ancha a
# proposito para no depender de que el canto de cada especie caiga justo
# en el pico -- el contraste se mantiene fuerte (24-29dB) en todo este
# rango. Recortada arriba de 9000 a 8000Hz porque la energia vocal real
# cae fuerte por encima de 8kHz sin ganar contraste a cambio.
TRIGGER_BANDA_MIN_HZ = 1500
TRIGGER_BANDA_MAX_HZ = 8000

# Percentil usado como "piso de ruido" propio de cada ventana de analisis
# (no un valor absoluto -- se adapta al ruido de fondo real del momento) y
# cuantos dB por encima de ese piso hace falta para contar como actividad.
TRIGGER_PERCENTIL_PISO = 20
TRIGGER_MARGEN_DB = 8

# --- Acumulacion del evento completo ---
# Cuanto silencio sostenido (sin actividad) hace falta para decidir que el
# canto termino. Mas alto = menos riesgo de cortar un canto con pausas
# naturales entre frases, a costa de un poco mas de latencia/audio de mas.
# Calibrado el 24/08 midiendo la distribucion real de pausas internas
# dentro de cantos (195 especies de Xeno-canto + muestra de campo real):
# con 1.0s, ~5-6% de las pausas internas superaban el umbral (riesgo real
# de partir un canto en dos eventos); con 1.5s baja a ~2.5-3.2%, casi la
# mitad, a costa de un poco mas de audio de cola por evento (barato, sin
# restriccion de memoria).
SILENCIO_FIN_EVENTO_S = 1.5

# Cuanto de ese silencio de espera queda realmente guardado al final del
# audio exportado, como aire despues del ultimo sonido real (el resto se
# recorta) -- porcentaje de SILENCIO_FIN_EVENTO_S, no un numero fijo, para
# que se reescale solo si se cambia el valor de arriba.
MARGEN_FINAL_PCT = 25

# Tope de seguridad: ningun evento se deja crecer mas que esto, pase lo
# que pase. Calibrado con la distribucion real de duracion de canto de 195
# especies (22/08): cubre incluso a las especies mas exigentes (Haematopus
# palliatus con p90 de ~22s) con margen.
DURACION_MAXIMA_EVENTO_S = 30

# Peso del piso de referencia "viejo" en el promedio movil que lo actualiza
# bloque a bloque mientras dura un evento (1.0 = nunca se actualiza, queda
# congelado como el 23/08; mas bajo = se adapta mas rapido a un ambiente
# que sube y se sostiene, ej. viento o trafico). Barrido el 24/08 con
# reproduccion sintetica: 0.8 recupera en ~3 bloques (~15s) ante un
# ambiente 4x mas alto sostenido, sin cortar de mas ningun canto real
# probado (con pausas internas cortas, todas por debajo de
# SILENCIO_FIN_EVENTO_S) -- el percentil bajo de TRIGGER_PERCENTIL_PISO ya
# protege esos casos, asi que valores mas bajos de alpha no mostraron
# riesgo adicional en las pruebas, pero se dejo un margen conservador.
ALPHA_PISO_EMA = 0.8

# Cuantos bloques de audio "anteriores" mantener siempre disponibles en el
# buffer rodante (sobrescribiendose), para poder reconstruir el inicio real
# de un canto aunque el disparo haya caido en medio de el, o cuando el
# evento sigue creciendo y cruza al bloque siguiente.
BLOQUES_BUFFER_ANTERIORES = 2

# --- Captura de audio en vivo (scripts/motor.py) ---
# Mismos parametros que usa birdnet_recording.sh de BirdNET-Pi para el
# mismo dispositivo (ver /etc/birdnet/birdnet.conf): tarjeta ALSA (o
# "default" para dejar que PulseAudio resuelva) y cantidad de canales.
REC_CARD = default
CHANNELS = 2

# Duracion de cada bloque leido de arecord antes de pasarlo a
# AcumuladorEventos.procesar_bloque(). No tiene relacion con
# DURACION_MAXIMA_EVENTO_S ni con el bloque de 15s de BirdNET-Pi -- el
# acumulador ya cruza bloques sin problema, asi que un bloque chico solo
# implica un piso de ruido (TRIGGER_PERCENTIL_PISO) mas localizado y menos
# latencia hasta que se evalua si arranco un evento.
DURACION_BLOQUE_S = 5
